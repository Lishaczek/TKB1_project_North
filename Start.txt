import os
import tkinter as tk
from scenes import SCENES

class Game:
    def __init__(self, root):
        self.root = root
        self.root.title("North")
        self.root.geometry("1100x700")

        self.image_cache = {}

        self.build_ui()
        self.show_scene("plan")

    def build_ui(self):
        # TOP (obrázek)
        self.top_frame = tk.Frame(self.root, bg="black", height=360)
        self.top_frame.pack(fill="x")
        self.top_frame.pack_propagate(False)

        self.scene_label = tk.Label(self.top_frame, bg="black")
        self.scene_label.pack(fill="both", expand=True)

        # BOTTOM
        self.bottom = tk.Frame(self.root)
        self.bottom.pack(fill="both", expand=True)

        # TEXT
        self.text_frame = tk.Frame(self.bottom)
        self.text_frame.pack(side="left", fill="both", expand=True)

        self.text = tk.Label(
            self.text_frame,
            justify="left",
            anchor="nw",
            font=("Consolas", 12),
            padx=10,
            pady=10,
            wraplength=700
        )
        self.text.pack(fill="both", expand=True)

        # VOLBY
        self.choice_frame = tk.Frame(self.bottom, width=250)
        self.choice_frame.pack(side="right", fill="y")
        self.choice_frame.pack_propagate(False)

    def load_image(self, name):
        if name in self.image_cache:
            return self.image_cache[name]

        path = os.path.join("images", name)

        if not os.path.exists(path):
            return None

        try:
            img = tk.PhotoImage(file=path)
        except tk.TclError:
            return None

        self.image_cache[name] = img
        return img

    def show_scene(self, scene_id):
        scene = SCENES[scene_id]

        # obrázek
        img = self.load_image(scene["image"])
        if img:
            self.scene_label.config(image=img)
            self.scene_label.image = img
        else:
            self.scene_label.config(text="[chybí obrázek]", fg="white")

        # text
        self.text.config(text=scene["text"])

        # volby
        for w in self.choice_frame.winfo_children():
            w.destroy()

        for label, next_scene in scene["choices"]:
            btn = tk.Button(
                self.choice_frame,
                text=label,
                command=lambda s=next_scene: self.show_scene(s)
            )
            btn.pack(fill="x", pady=5, padx=5)

if __name__ == "__main__":
    root = tk.Tk()
    Game(root)
    root.mainloop()
